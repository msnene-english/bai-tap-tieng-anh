const quizData = buildQuizData();
// ===== SHUFFLE =====
function shuffleArray(arr) {
let a = [...arr];
for (let i = a.length - 1; i > 0; i--) {
const j = Math.floor(Math.random() * (i + 1));
[a[i], a[j]] = [a[j], a[i]];
}
return a;
}

let totalQ = quizData.length;
let answered = 0;
let correctCount = 0;

function updateScore() {
document.getElementById('scoreDisplay').textContent = correctCount + ' / ' + answered;

document.getElementById('progressBar').style.width = pct + '%';
}
function buildQuiz() {
const container = document.getElementById('quizContainer');
quizData.forEach((data, idx) => {
const shuffledOpts = shuffleArray(data.options);
const card = document.createElement('div');
card.className = 'q-card';
let optHTML = shuffledOpts.map((o) => `
<button class="opt-btn" data-correct="${o.correct}" onclick="checkMCQ(this, event, ${idx})">
${o.text}
</button>
`).join('');
card.innerHTML = `
<div class="q-num">Câu ${data.num}</div>
<div class="q-text">❓ ${data.question}</div>
<div class="options">${optHTML}</div>
<div class="explanation" id="expl_${idx}">
<div class="expl-title">💡 Giải thích siêu chi tiết</div>
${data.explanation}
</div>
`;
container.appendChild(card);
});
}

window.checkMCQ = function(btn, evt, idx) {
if (ansState[idx]) return;
ansState[idx] = true;
const card = btn.closest('.q-card');
const allBtns = card.querySelectorAll('.opt-btn');
allBtns.forEach(b => b.disabled = true);
const isCorrect = btn.dataset.correct === 'true';
if (isCorrect) {
btn.classList.add('correct');
playDing();
correctCount++;
spawnConfetti(evt.clientX, evt.clientY);
} else {
btn.classList.add('wrong');
playBuzz();
allBtns.forEach(b => {
if (b.dataset.correct === 'true') {
b.classList.add('correct');
b.style.opacity = '0.7';
}
});
}
answered++;
updateScore();
const expl = document.getElementById('expl_' + idx);
if(expl) expl.classList.add('show');
setTimeout(() => expl.scrollIntoView({behavior:'smooth', block:'center'}), 200);
}
